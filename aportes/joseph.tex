\begin{frame}[fragile,t]{Ejemplo del intercambio XML}

\begin{columns}[T,onlytextwidth]
\begin{column}{.49\textwidth}
{\color{acento}\bfseries Solicitud}\\[2mm]
{\scriptsize
\begin{verbatim}
<soap:Envelope
 xmlns:soap=
 "http://www.w3.org/2003/05/soap-envelope"
 xmlns:tit=

  <soap:Body>
    <tit:ConsultarRegistro>
      <tit:codigoRegistro>
        TEST-001
      </tit:codigoRegistro>
    </tit:ConsultarRegistro>
  </soap:Body>
</soap:Envelope>
\end{verbatim}
}
\end{column}

\begin{column}{.49\textwidth}

\medskip
\kgnota{Nombres y valores ficticios. SOAP intercambia XML \cite{ayfan2024soap}.}
\note{85 segundos. Recorrer la solicitud: el SSDC prepara un mensaje XML con los datos que exigiría el servicio. El proveedor procesaría la consulta y devolvería otro mensaje XML con el resultado. TEST-001 y VALIDO son valores de ejemplo, no datos ni respuestas de un proveedor real.}
\end{frame}


\begin{frame}[t]{Demora, caída o error del servicio}
\tema{Joseph Calderón · Detección y respuesta a fallos}

\begin{columns}[T,onlytextwidth]
\begin{column}{.32\textwidth}
{\color{ambar}\bfseries Tiempo de espera}\\[2mm]
Si no llega una respuesta dentro del límite configurado, registrar la demora e informar que el proveedor no respondió.
\end{column}

\begin{column}{.32\textwidth}
{\color{ambar}\bfseries Fallo de conexión}\\[2mm]
Si no se logra conectar, informar que no fue posible completar la consulta.
\end{column}

\begin{column}{.32\textwidth}
{\color{ambar}\bfseries SOAP Fault}\\[2mm]
Si llega un Fault, revisar el error devuelto y comunicar que la consulta no pudo procesarse.
\end{column}
\end{columns}

\medskip
\colorbox{suave}{\parbox{.96\linewidth}{
\smallskip
\textbf{\color{acento}Regla para el SSDC:}
la falta de respuesta no demuestra que el título sea inválido.
\smallskip
}}

\medskip
\kgnota{Manejo propuesto; todavía no está implementado ni probado en el SSDC.}
\note{60 segundos. Distinguir entre demora, fallo de conexión y SOAP Fault. En los dos primeros casos no se recibe una respuesta válida; en el tercero, el servicio sí responde con información del error. El SSDC informaría el problema y no marcaría el título como inválido por falta de respuesta.}
\end{frame}
